cmake_minimum_required(VERSION 3.20)
project(vore_nx_host VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vore_cartridge source/cartridge.cpp source/sha256.cpp)
target_include_directories(vore_cartridge PUBLIC include)
if(MSVC)
  target_compile_definitions(vore_cartridge PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(vore_cartridge PRIVATE /W4 /WX)
else()
  target_compile_options(vore_cartridge PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(vore-rom source/host_main.cpp)
target_link_libraries(vore-rom PRIVATE vore_cartridge)
enable_testing()
add_executable(vore-tests tests/core_tests.cpp)
target_link_libraries(vore-tests PRIVATE vore_cartridge)
add_test(NAME cartridge-and-hash COMMAND vore-tests)
add_executable(vore-video-tests tests/runtime_video_tests.cpp)
target_include_directories(vore-video-tests PRIVATE include)
add_test(NAME runtime-video COMMAND vore-video-tests)
add_executable(vore-local-tests tests/local_rom_tests.cpp)
target_link_libraries(vore-local-tests PRIVATE vore_cartridge)
